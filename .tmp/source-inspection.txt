import 'server-only'

import config from '@payload-config'
import { cookies } from 'next/headers'
import { getPayload } from 'payload'

import { auth } from '@/auth'
import type { User } from '@/payload-types'

const APP_SESSION_COOKIE = 'app-session'

export function createAppAuth(collection: 'users') {
  return {
    async login(email: string, password: string) {
      const payload = await getPayload({ config })
      const result = await payload.login({
        collection,
        data: { email, password },
      })

      if (!result.token || !result.user || result.user.active === false) {
        return null
      }

      const cookieStore = await cookies()
      cookieStore.set(APP_SESSION_COOKIE, result.token, {
        httpOnly: true,
        maxAge: 60 * 60 * 8,
        path: '/',
        sameSite: 'lax',
        secure: process.env.NODE_ENV === 'production',
      })

      return result.user
    },

    async logout() {
      const cookieStore = await cookies()
      cookieStore.delete(APP_SESSION_COOKIE)
    },

    async currentUser(): Promise<User | null> {
      const oauthSession = await auth()
      const oauthUserId = oauthSession?.user?.payloadUserId
      if (oauthUserId) {
        const payload = await getPayload({ config })
        const oauthUser = await payload.findByID({
          collection,
          id: oauthUserId,
          overrideAccess: true,
        })
        if (oauthUser.active === false || oauthUser.removedAt) return null
        return oauthUser
      }

      const token = (await cookies()).get(APP_SESSION_COOKIE)?.value
      if (!token) return null

      const payload = await getPayload({ config })
      const authHeaders = new Headers({ Authorization: `JWT ${token}` })
      const { user } = await payload.auth({ headers: authHeaders })

      if (!user || user.collection !== collection || user.active === false || user.removedAt) return null
      return user
    },
  }
}

export const appAuth = createAppAuth('users')

import config from '@payload-config'
import { randomBytes } from 'node:crypto'
import { getPayload } from 'payload'

import type { User } from '@/payload-types'

export const PRIMARY_ADMIN_EMAIL = 'baliwangmarkjhon@gmail.com'

export type VerifiedGoogleIdentity = {
  providerId: string
  email: string
  name: string
  image?: string | null
  emailVerified: boolean
}

export async function syncVerifiedGoogleUser(identity: VerifiedGoogleIdentity): Promise<User> {
  const email = identity.email.trim().toLowerCase()
  if (!identity.emailVerified || !identity.providerId || !email) {
    throw new Error('Google did not return a verified email identity.')
  }

  const payload = await getPayload({ config })
  const byProvider = await payload.find({
    collection: 'users',
    where: { googleProviderId: { equals: identity.providerId } },
    limit: 1,
    overrideAccess: true,
  })
  const byEmail =
    byProvider.docs[0] ??
    (
      await payload.find({
        collection: 'users',
        where: { email: { equals: email } },
        limit: 1,
        overrideAccess: true,
      })
    ).docs[0]
  const role = email === PRIMARY_ADMIN_EMAIL ? 'admin' : (byEmail?.role ?? 'guest')

  if (byEmail) {
    return payload.update({
      collection: 'users',
      id: byEmail.id,
      overrideAccess: true,
      data: {
        name: identity.name || byEmail.name,
        role,
        active: true,
        googleProviderId: identity.providerId,
        avatarUrl: identity.image ?? byEmail.avatarUrl,
      },
    })
  }

  return payload.create({
    collection: 'users',
    overrideAccess: true,
    data: {
      name: identity.name || email.split('@')[0],
      email,
      password: randomBytes(32).toString('base64url'),
      role,
      active: true,
      googleProviderId: identity.providerId,
      avatarUrl: identity.image ?? undefined,
    },
  })
}

export type AuthActionResult = { ok: boolean; message: string }

const destinationForRole = (role: 'guest' | 'staff' | 'admin') =>
  role === 'guest' ? '/' : '/operations'

export async function loginAction(
  _previous: AuthActionResult,
  formData: FormData,
): Promise<AuthActionResult> {
  const email = String(formData.get('email') ?? '').trim().toLowerCase()
  const password = String(formData.get('password') ?? '')
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || password.length < 8) {
    return { ok: false, message: 'Enter a valid email and password.' }
  }
  try {
    const user = await appAuth.login(email, password)
    if (!user) return { ok: false, message: 'This account is inactive.' }
    redirect(destinationForRole(user.role))
  } catch (error) {
    if (error && typeof error === 'object' && 'digest' in error) throw error
    return { ok: false, message: 'Email or password is incorrect.' }
  }
}

export async function signUpAction(
  _previous: AuthActionResult,
  formData: FormData,
): Promise<AuthActionResult> {
  const name = String(formData.get('name') ?? '').trim()
  const email = String(formData.get('email') ?? '').trim().toLowerCase()
  const password = String(formData.get('password') ?? '')
  const confirmPassword = String(formData.get('confirmPassword') ?? '')
  if (name.length < 2) return { ok: false, message: 'Enter your full name.' }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return { ok: false, message: 'Enter a valid email address.' }
  }
  if (password.length < 8 || !/[A-Za-z]/.test(password) || !/\d/.test(password)) {
    return { ok: false, message: 'Use at least 8 characters with a letter and number.' }
  }
  if (password !== confirmPassword) return { ok: false, message: 'Passwords do not match.' }

  try {
    const payload = await getPayload({ config })
    await payload.create({
      collection: 'users',
      draft: false,
      overrideAccess: false,
      data: { name, email, password, role: 'guest', active: true },
    })
    const user = await appAuth.login(email, password)
    if (!user) return { ok: false, message: 'Your account was created, but sign-in failed.' }
    redirect('/')
  } catch (error) {
    if (error && typeof error === 'object' && 'digest' in error) throw error
    return { ok: false, message: 'An account with this email may already exist.' }
  }
}

export async function removeStaffAction(id: string): Promise<ActionResult> {
  try {
    const user = await requireAdmin()
    if (String(user.id) === id) {
      return { ok: false, message: 'You cannot remove your own administrator account.' }
    }
    const payload = await getPayload({ config })
    const member = await payload.findByID({
      collection: 'users',
      id,
      overrideAccess: false,
      user,
    })
    if (member.role !== 'staff') {
      return { ok: false, message: 'Only staff accounts can be removed here.' }
    }
    await payload.update({
      collection: 'users',
      id,
      overrideAccess: false,
      user,
      data: { active: false, removedAt: new Date().toISOString() },
    })
    return { ok: true, message: 'Staff access removed. Historical activity was preserved.' }
  } catch {
    return { ok: false, message: 'The staff account could not be removed.' }
  }
}

export async function updateRateAction(input: {

    await payload.update({

    })

}

          <section className="mt-7 rounded-2xl border border-[#dce7e2] bg-white p-6">

            <p className="mt-1 text-sm text-[#647b75]">Update published nightly rates and review room capacity.</p>
            <div className="mt-6 grid gap-4 lg:grid-cols-2">
              {rates.map((rate) => (
                <article key={rate.id} className="flex flex-col gap-4 rounded-xl border border-[#dce7e2] p-4 sm:flex-row sm:items-center">
                  <div className="flex-1"><p className="font-semibold">{rate.name}</p><p className="mt-1 text-sm text-[#647b75]">Up to {rate.capacity} guests · {rate.active ? 'Active inventory' : 'Inactive'}</p></div>
                  <form className="flex gap-2" action={(formData) => run(() => updateRateAction({ id: rate.id, rate: Number(formData.get('rate')) }))}>
                    <input name="rate" type="number" min="0" defaultValue={rate.rate} className="min-h-11 w-32 rounded-lg border border-[#cfddd8] px-3" aria-label={`Rate for ${rate.name}`} />
                    <button disabled={pending} className="min-h-11 rounded-lg bg-[#123d33] px-4 font-semibold text-white">Save</button>
                  </form>
                </article>
              ))}
            </div>
          </section>
        )}
      </div>
    </main>
  )
